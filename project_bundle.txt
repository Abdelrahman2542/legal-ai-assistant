### 1. Create and Activate the Virtual Environment (venv)
```bash
python -m venv .venv
.venv\Scripts\activate
```
### 1. Install the required packages
```bash
cd src
pip install -r requirements.txt

```
### Setup the environment variables
```bash
copy .env.example .env
```
Set your environment variables in the .env file. Like OPENAI_API_KEY value

### Run fast api server
```bash
uvicorn main:app --reload
```

import os
import json

# تحديد المسارات حسب هيكل مشروعك
BASE_DIR = os.path.dirname(os.path.dirname(os.path.dirname(os.path.abspath(__file__))))
DATA_DIR = os.path.join(BASE_DIR, "data")

STAGED_DIR = os.path.join(DATA_DIR, "staging", "labor law")
PROCESS_DIR = os.path.join(DATA_DIR, "processed", "labor law")

def chunk_labor_law():
    input_path = os.path.join(STAGED_DIR, "extracted_clean.txt")
    
    print(f"🔍 البحث عن ملف الـ staged في مسار: {input_path}")
    
    if not os.path.exists(input_path):
        print(f"❌ خطأ: ملف الـ staged غير موجود في هذا المسار: {input_path}")
        print("💡 من فضلك شغّل cleaner.py أولاً لاستخراج النص الخام.")
        return

    # التأكد من وجود مجلد processed/labor law
    os.makedirs(PROCESS_DIR, exist_ok=True)

    print("📖 جاري قراءة النص المستخرج وتقسيمه إلى مواد بالطريقة الذكية...")
    with open(input_path, "r", encoding="utf-8") as f:
        full_text = f.read()

    lines = full_text.split('\n')

    chunks = []
    current_article_title = "مقدمة"
    current_article_content = ""
    article_counter = 1

    for line in lines:
        stripped_line = line.strip()

        # شروط بداية المادة بدقة للتأكد من التقسيم السليم
        is_new_article = (
            stripped_line.startswith("مادة") or
            stripped_line.startswith("المادة") or
            stripped_line.startswith("ﻣﺎﺩﺓ") or
            stripped_line.startswith("ﺍﳌﺎﺩﺓ") or
            stripped_line.startswith("مادة)") or
            stripped_line.startswith("ﻣﺎﺩﺓ)")
        )

        if is_new_article:
            # لو فيه مادة قديمة متجمعة، نحفظها كـ Dict منظم
            if current_article_content.strip():
                chunk_data = {
                    "chunk_id": f"labor_art_{article_counter}",
                    "article_title": current_article_title,
                    "section": "issuance" if "إصدار" in current_article_content[:200] else "main_law",
                    "content": current_article_content.strip()
                }
                chunks.append(chunk_data)
                article_counter += 1

            # نبدأ مادة جديدة
            current_article_title = stripped_line
            current_article_content = ""
        else:
            # نجمع محتوى السطور جوه نفس المادة الحالية
            current_article_content += line + "\n"

    # إضافة آخر مادة في الملف
    if current_article_content.strip():
        chunk_data = {
            "chunk_id": f"labor_art_{article_counter}",
            "article_title": current_article_title,
            "section": "issuance" if "إصدار" in current_article_content[:200] else "main_law",
            "content": current_article_content.strip()
        }
        chunks.append(chunk_data)

    # حفظ النتائج في ملف JSON
    output_json_path = os.path.join(PROCESS_DIR, "chunks.json")
    with open(output_json_path, "w", encoding="utf-8") as f:
        json.dump(chunks, f, ensure_ascii=False, indent=4)

    print(f"✅ تم بنجاح تقسيم القانون إلى {len(chunks)} مادة/تشانك!")
    print(f"💾 تم حفظ التشانكات في: {output_json_path}")

if __name__ == "__main__":
    chunk_labor_law()

import os
import json
import fitz  # PyMuPDF

# تحديد مسار جذر المشروع بثبات (الوصول لمجلد legal-ai-assistant مباشرة)
BASE_DIR = os.path.abspath(os.path.join(os.path.dirname(__file__), "..", ".."))
DATA_DIR = os.path.join(BASE_DIR, "data")

RAW_DIR = os.path.join(DATA_DIR, "raw", "labor law")
STAGED_DIR = os.path.join(DATA_DIR, "staging")
CORRECTIONS_FILE = os.path.join(DATA_DIR, "corrections.json")

def load_corrections():
    """تحميل قاموس التصحيحات الإملائية لو ملف corrections.json موجود"""
    if os.path.exists(CORRECTIONS_FILE):
        try:
            with open(CORRECTIONS_FILE, "r", encoding="utf-8") as f:
                return json.load(f)
        except Exception as e:
            print(f"تحذير: لم يتم قراءة ملف corrections.json بشكل صحيح: {e}")
    return {}

def clean_text(text: str, corrections: dict) -> str:
    """عمليات التنظيف الأساسية وتطبيق التصحيحات"""
    for wrong, right in corrections.items():
        text = text.replace(wrong, right)
    return text

def extract_and_stage_pdf(pdf_filename="قانون العمل.pdf"):
    pdf_path = os.path.join(RAW_DIR, pdf_filename)
    
    if not os.path.exists(pdf_path):
        print(f"خطأ: ملف الـ PDF غير موجود في المسار: {pdf_path}")
        print(f"تأكد أن الملف موجود في: {pdf_path}")
        return
    
    # التأكد من وجود مجلد staged
    os.makedirs(STAGED_DIR, exist_ok=True)
    
    print(f"جاري فتح واستخراج النص من: {pdf_path} ...")
    doc = fitz.open(pdf_path)
    full_text = ""
    
    for page_num, page in enumerate(doc):
        page_text = page.get_text("text")
        full_text += f"\n--- الصفحة رقم {page_num + 1} ---\n" + page_text

    # تحميل التصحيحات وتطبيقها
    corrections = load_corrections()
    cleaned_full_text = clean_text(full_text, corrections)
    
    # حفظ الناتج في مجلد staged
    output_filename = "extracted_clean.txt"
    output_path = os.path.join(STAGED_DIR,"labor law", output_filename)
    
    with open(output_path, "w", encoding="utf-8") as f:
        f.write(cleaned_full_text)
        
    print(f"تم بنجاح! إجمالي الحروف المستخرجة: {len(cleaned_full_text)}")
    print(f"تم حفظ النص المنظف في: {output_path}")

if __name__ == "__main__":
    extract_and_stage_pdf()
